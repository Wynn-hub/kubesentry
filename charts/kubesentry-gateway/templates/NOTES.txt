  echo "http://$NODE_IP:{{ .Values.console.service.nodePort | default .Values.console.service.port }}"
{{- else }}

  kubectl -n {{ .Values.namespace }} port-forward svc/kubesentry-console {{ .Values.console.service.port }}:{{ .Values.console.service.port }}

Then open http://localhost:{{ .Values.console.service.port }}
{{- end }}
{{- end }}
